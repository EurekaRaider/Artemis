cmake_minimum_required(VERSION 3.24)
project(ArtemisComputerUse LANGUAGES CXX)
add_executable(artemis-computer-use main.cpp transport.cpp applications.cpp automation.cpp observation.cpp actions.cpp capture.cpp input-control.cpp)
target_compile_features(artemis-computer-use PRIVATE cxx_std_20)
target_compile_definitions(artemis-computer-use PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
set_property(TARGET artemis-computer-use PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_compile_options(artemis-computer-use PRIVATE /W4 /EHsc /permissive-)
target_link_libraries(artemis-computer-use PRIVATE windowsapp ole32 oleaut32 uiautomationcore user32 shell32 propsys dwmapi d3d11 dxgi windowscodecs bcrypt crypt32 advapi32)
target_link_options(artemis-computer-use PRIVATE /MANIFEST:EMBED /MANIFESTINPUT:helper.manifest)
configure_file(helper.manifest helper.manifest COPYONLY)
if(DEFINED ARTEMIS_NODE_HEADERS AND DEFINED ARTEMIS_NODE_LIBRARY)
  file(TO_CMAKE_PATH "${ARTEMIS_NODE_DELAY_HOOK}" ARTEMIS_NODE_DELAY_HOOK)
  file(TO_CMAKE_PATH "${ARTEMIS_NODE_HEADERS}" ARTEMIS_NODE_HEADERS)
  file(TO_CMAKE_PATH "${ARTEMIS_NODE_LIBRARY}" ARTEMIS_NODE_LIBRARY)
  add_library(artemis-computer-preview MODULE ../preview/windows.cpp "${ARTEMIS_NODE_DELAY_HOOK}")
  target_compile_features(artemis-computer-preview PRIVATE cxx_std_20)
  target_compile_definitions(artemis-computer-preview PRIVATE NAPI_VERSION=8 UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN HOST_BINARY="node.exe")
  target_include_directories(artemis-computer-preview PRIVATE "${ARTEMIS_NODE_HEADERS}")
  target_compile_options(artemis-computer-preview PRIVATE /W4 /EHsc /permissive-)
  set_property(TARGET artemis-computer-preview PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
  set_target_properties(artemis-computer-preview PROPERTIES PREFIX "" SUFFIX ".node")
  target_link_libraries(artemis-computer-preview PRIVATE "${ARTEMIS_NODE_LIBRARY}" delayimp windowsapp d3d11 dxgi user32 runtimeobject)
  target_link_options(artemis-computer-preview PRIVATE /DELAYLOAD:node.exe)
endif()
