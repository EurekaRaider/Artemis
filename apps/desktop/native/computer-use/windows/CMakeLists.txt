cmake_minimum_required(VERSION 3.24)
project(ArtemisComputerUse LANGUAGES CXX)
add_executable(artemis-computer-use main.cpp transport.cpp applications.cpp automation.cpp observation.cpp actions.cpp capture.cpp input-control.cpp)
target_compile_features(artemis-computer-use PRIVATE cxx_std_20)
target_compile_definitions(artemis-computer-use PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
set_property(TARGET artemis-computer-use PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
target_compile_options(artemis-computer-use PRIVATE /W4 /EHsc /permissive-)
target_link_libraries(artemis-computer-use PRIVATE windowsapp ole32 oleaut32 uiautomationcore user32 shell32 propsys dwmapi d3d11 dxgi windowscodecs bcrypt crypt32 advapi32)
target_link_options(artemis-computer-use PRIVATE /MANIFEST:EMBED /MANIFESTINPUT:helper.manifest)
configure_file(helper.manifest helper.manifest COPYONLY)
